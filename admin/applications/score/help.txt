# applications/quiz/services.py
from applications.quiz.models import Quiz, QuizAttempt
from applications.score import api as score_api


def submit_quiz_service(user_id: int, quiz_id: int, answer_index: int) -> dict:
    quiz = Quiz.objects.filter(id=quiz_id, is_active=True).first()
    if not quiz:
        return {"ok": False, "message": "کوئیز یافت نشد"}

    # ۱) ذخیره پاسخ (مسئولیت اپ quiz)
    correct = answer_index == quiz.correct_index
    attempt = QuizAttempt.objects.create(
        user_id=user_id, quiz=quiz,
        answer_index=answer_index, is_correct=correct,
    )

    # ۲) امتیاز (عوارض جانبی — بعد از ذخیره)
    log = score_api.award(
        user_id,
        code="DAILY_QUIZ",
        amount_override=1 if correct else 0,   # غلط هم ۰ ثبت می‌شه
        reference_type="quiz_attempt",
        reference_id=str(attempt.id),
        reason="کوئیز روزانه",
        extra={"quiz_id": quiz.id, "correct": correct},
    )

    return {
        "ok": True,
        "correct": correct,
        "awarded": log is not None,
        "score_log_id": log.id if log else None,
    }

from applications.gift.models import GiftOrder
from applications.score import api as score_api
from applications.score.services import ScoreError


def create_gift_order(user, gift, quantity: int = 1) -> GiftOrder:
    # ۱) ساخت سفارش
    order = GiftOrder.objects.create(
        user=user, gift=gift, quantity=quantity,
        score_cost=gift.score_cost * quantity,
        status="pending",
    )

    # ۲) کسر امتیاز — اگه موجودی کافی نباشه، ScoreError می‌ده
    try:
        log = score_api.spend(
            user, "GIFT_REDEEM",
            amount=order.score_cost,
            reference_type="gift_order", reference_id=str(order.id),
            reason=f"خرید هدیه #{order.id}",
            extra={"gift_id": gift.id, "quantity": quantity},
        )
        order.score_log_id = log.id
        order.status = "approved"
        order.save(update_fields=["score_log_id", "status"])
        return order
    except ScoreError:
        order.delete()
        raise


def cancel_gift_order(order_id: int) -> None:
    order = GiftOrder.objects.filter(id=order_id).first()
    if not order or order.status == "cancelled":
        return

    order.status = "cancelled"
    order.save(update_fields=["status"])

    # ✅ برگشت امتیاز → ORDER_REFUND
    if order.score_log_id:
        score_api.refund(
            order.score_log_id,
            reason=f"لغو سفارش هدیه #{order.id}",
        )

@router.post("/verify/")
async def verify_mobile(form: dict = Depends(DependencyDataRequest)):
    mobile = form.get("mobile")
    code = form.get("code")

    # ── تأیید کد ──────────────────────────────────
    ok = await sync_to_async(_verify_code)(mobile, code)
    if not ok:
        return error_response("کد نامعتبر است")

    # ── فعال‌سازی کاربر ───────────────────────────
    user = await sync_to_async(_activate_user)(mobile)

    # ── REFERRAL (فقط اینجا!) ─────────────────────
    await sync_to_async(reward_referral_after_verification)(user)

    return success_response("شماره تأیید شد")


@sync_to_async
def _activate_user(mobile):
    user = User.objects.get(mobile=mobile)
    user.is_mobile_verified = True
    user.save(update_fields=["is_mobile_verified"])
    return user



# applications/user/services.py
def reward_referral_after_verification(user):
    """بعد از verify موبایل کاربر دعوت‌شده صدا زده می‌شه."""
    from applications.score import api as score_api

    with transaction.atomic():
        inv = (
            UserInvitation.objects
            .select_for_update()
            .filter(invited_user=user, is_rewarded=False)
            .first()
        )
        if not inv:
            return

        score_api.award(
            inv.inviter, "REFERRAL",
            reference_type="invitation",
            reference_id=f"{inv.id}:inviter",
            reason=f"دعوت {user.mobile}",
            extra={"role": "inviter"},
        )
        score_api.award(
            user, "REFERRAL",
            reference_type="invitation",
            reference_id=f"{inv.id}:invited",
            reason="هدیه ثبت‌نام با کد دعوت",
            extra={"role": "invited"},
        )

        inv.is_rewarded = True
        inv.rewarded_at = timezone.now()
        inv.save(update_fields=["is_rewarded", "rewarded_at"])

def checkin(user) -> dict:

        reference_type="checkin",
        reference_id=f"{user.id}:{now.date().isoformat()}",
        reason="حضور در قرار جمعه",
    )
    return {"ok": True, "awarded": log is not None}

from applications.score.services import ScoreError

def create_gift_order(user, gift, quantity: int = 1) -> GiftOrder:
    cost = gift.score_cost * quantity

    # ۱) اول کسر امتیاز (اگه موفق شد، سفارش ساخته می‌شه)
    code = (
        UserScoreLog.TransactionCode.GIFT_REDEEM
        if gift.kind == "showcase"
        else UserScoreLog.TransactionCode.CUSTOM_GIFT
    )

    # ابتدا یه order موقت می‌سازیم تا reference_id داشته باشیم
    order = GiftOrder.objects.create(
        user=user, gift=gift, quantity=quantity,
        score_cost=cost, status="pending",
    )

    try:
        log = score_api.spend(   # ← SPENT (throw می‌کنه اگه موجودی نباشه)
            user,
            code=code,
            amount=cost,
            reference_type="gift_order",
            reference_id=str(order.id),
            reason=f"خرید هدیه #{order.id}",
            extra={"gift_id": gift.id, "quantity": quantity},
        )
    except ScoreError:
        order.delete()   # rollback دستی
        raise

    order.score_log_id = log.id
    order.status = "approved"
    order.save(update_fields=["score_log_id", "status"])
    return order


def cancel_gift_order(order_id: int, reason: str = "") -> None:
    order = GiftOrder.objects.filter(id=order_id).first()
    if not order or order.status == "cancelled":
        return

    order.status = "cancelled"
    order.save(update_fields=["status"])

    # 24) برگشت امتیاز → ORDER_REFUND
    if order.score_log_id:
        score_api.refund(
            order.score_log_id,
            reason=reason or f"لغو سفارش هدیه #{order.id}",
        )

def run_weekly_streak_job():
    """شنبه‌ها اجرا شود."""
    from applications.user.models import User
    now = timezone.now()
    week_start = (now - timedelta(days=7)).replace(hour=0, minute=0, second=0, microsecond=0)
    week_end = now.replace(hour=0, minute=0, second=0, microsecond=0)

    for user in User.objects.filter(is_active=True).iterator():
        days = (
            UserScoreLog.objects
            .filter(
                user_id=user.id,
                created_at__gte=week_start,
                created_at__lt=week_end,
            )
            .dates("created_at", "day")
            .count()
        )
        if days >= 7:
            try:
                score_api.award(
                    user,
                    code=UserScoreLog.TransactionCode.WEEKLY_STREAK,
                    reference_type="weekly_streak",
                    reference_id=f"{user.id}:{week_end.date().isoformat()}",
                    reason="تکمیل هفته با ۷ روز فعالیت",
                )
            except Exception:
                logger.exception("WEEKLY_STREAK failed user=%s", user.id)
                

def run_gallery_top_job():
    """شنبه‌ها اجرا شود."""
    try:
        from applications.gallery.models import GalleryPhoto
    except ImportError:
        return

    week_start = (timezone.now() - timedelta(days=7)).replace(
        hour=0, minute=0, second=0, microsecond=0,
    )
    winner = (
        GalleryPhoto.objects
        .filter(created_at__gte=week_start)
        .order_by("-likes_count")
        .first()
    )
    if winner:
        try:
            score_api.award(
                winner.user,
                code=UserScoreLog.TransactionCode.GALLERY_TOP,
                reference_type="gallery_photo",
                reference_id=str(winner.id),
                reason="عکس برتر هفته",
            )
        except Exception:
            logger.exception("GALLERY_TOP failed photo=%s", winner.id)


def run_expire_job():
    """روزانه اجرا شود."""
    from applications.score.models import ScoreRule, UserScore

    rule = ScoreRule.objects.filter(code="EXPIRE", is_active=True).first()
    if not rule:
        return
    days = rule.metadata.get("expire_after_days", 365)
    cutoff = timezone.now() - timedelta(days=days)

    recent_earners = UserScoreLog.objects.filter(
        direction=UserScoreLog.Direction.EARNED,
        created_at__gte=cutoff,
    ).values_list("user_id", flat=True)

    stale = UserScore.objects.filter(
        current_balance__gt=0,
    ).exclude(user_id__in=recent_earners)

    for score in stale.iterator():
        try:
            score_api.spend(
                score.user_id,
                code=UserScoreLog.TransactionCode.EXPIRE,
                amount=score.current_balance,
                reason=f"انقضای امتیاز پس از {days} روز بی‌فعالیتی",
                source=UserScoreLog.Source.SYSTEM,
            )
        except Exception:
            logger.exception("EXPIRE failed user=%s", score.user_id)

@router.post("/admin/adjust/")
async def admin_adjust(
    form: dict = Depends(DependencyDataRequest),
    admin: User = Depends(get_current_user),
):
    try:
        if not (admin.is_staff or admin.is_superuser):
            return error_response("دسترسی ادمین لازم است", status_code=403)

        target = await database_sync_to_async(
            lambda: User.objects.filter(uuid=form.get("user_uuid")).first()
        )()
        if not target:
            return error_response("کاربر یافت نشد", status_code=404)

        direction = form.get("direction")
        if direction not in ("earned", "spent"):
            return error_response("direction باید earned یا spent باشد")

        log = await database_sync_to_async(score_api.adjust)(
            target,
            int(form.get("amount", 0)),
            direction,
            admin=admin,
            reason=form.get("reason") or "تنظیم دستی",
            extra={"admin_name": admin.full_name},
        )
        return success_response("امتیاز تنظیم شد", _log(log))
    except PermissionError as e:
        return error_response(str(e), status_code=403)
    except ScoreError as e:
        return error_response(str(e))
